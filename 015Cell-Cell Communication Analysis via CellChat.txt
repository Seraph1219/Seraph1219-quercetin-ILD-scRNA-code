# Title: Cell-Type-Specific Ligand-Receptor Mediated Communication in ILD
#        and Hub Gene-Driven Signaling Networks

# Description:
#   This script performs CellChat-based intercellular communication analysis
#   using the fine-annotated Seurat object. It:

#   3. Visualizes global and pathway-specific chord diagrams

# --------------------- Section 1: Load Final Seurat Object --------------------
scRNA <- readRDS("08_Lung_Donor_vs_ILD_Final.rds")

# --------------------- Section 2: Build CellChat Metadata ---------------------
meta <- data.frame(
  cell      = colnames(scRNA),
  celltype  = scRNA$celltype_fine_corrected,
  condition = scRNA$Group,
  stringsAsFactors = FALSE
)
rownames(meta) <- meta$cell
meta$cell <- NULL

# Define cell subtypes to retain
keep_subtypes <- c(
  "iFib", "ImmFib", "mFib",
  "AM", "IM", "InfMac", "MtMac", "LAMac",
  "SPP1+Mac", "IFNMac", "FibMac", "MetMac", "CTLMac", "MDM",
  "VenEC", "ArtEC", "laEC", "SaEC", "MLEC", "LEC",
  "CapEC-II", "Cap. EC", "BSEC",
  "IREC", "AT1", "BATEC", "MSEC",
  "CC", "CEC", "SEC", "ASEC", "Club ", "SecEpi", "CilEpi",
  "DC",
  "B", "B cell", "Plasma"
)

meta <- meta[meta$celltype %in% keep_subtypes, ]

# --------------------- Section 3: Define Cell Type Ordering -------------------
# Order determines adjacency in chord diagrams (grouped by major lineage)
group_order <- c(
  # Macrophage
  "AM", "IM", "InfMac", "MtMac", "LAMac",
  "SPP1+Mac", "IFNMac", "FibMac", "MetMac", "CTLMac", "MDM",
  # Fibroblast
  "iFib", "ImmFib", "mFib",
  # Endothelium
  "VenEC", "ArtEC", "laEC", "SaEC", "MLEC", "LEC",
  "CapEC-II", "Cap. EC", "BSEC",
  # Epithelium
  "AT1", "IREC", "BATEC", "MSEC",
  "CC", "CEC", "SEC", "ASEC", "Club ", "SecEpi", "CilEpi",
  # Mono / DC
  "DC",
  # B / Plasma
  "B", "B cell", "Plasma"
)

group_order <- group_order[group_order %in% meta$celltype]
meta$celltype  <- factor(meta$celltype, levels = group_order)
meta$condition <- factor(meta$condition)

cat("Cell type distribution:\n")
print(table(meta$celltype))
cat("Condition distribution:\n")
print(table(meta$condition))

# --------------------- Section 4: Extract Expression Matrix -------------------
expr <- GetAssayData(scRNA, assay = "RNA", layer = "data")
expr <- expr[, rownames(meta), drop = FALSE]

# --------------------- Section 5: Build CellChat Objects by Condition ---------
conds <- unique(meta$condition)
conds <- conds[!is.na(conds)]
conds <- trimws(conds)

cellchat_list <- lapply(conds, function(cond) {

  cat("\n==============================\n")
  cat("Processing:", cond, "\n")

  idx <- meta$condition == cond
  meta_sub  <- meta[idx, , drop = FALSE]
  expr_sub  <- expr[, idx, drop = FALSE]

  if (nrow(meta_sub) == 0 || ncol(expr_sub) == 0) {
    cat("  No cells found for", cond, "- skipping\n")
    return(NULL)
  }

  # Re-level celltype factor for this subset
  meta_sub$celltype <- factor(
    meta_sub$celltype,
    levels = group_order[group_order %in% meta_sub$celltype]
  )

  print(table(meta_sub$celltype))
  colnames(expr_sub) <- rownames(meta_sub)

  cc <- createCellChat(
    object   = expr_sub,
    meta     = meta_sub,
    group.by = "celltype"
  )

  cc@DB <- CellChatDB.human

  cc <- subsetData(cc)
  cc <- identifyOverExpressedGenes(cc)
  cc <- identifyOverExpressedInteractions(cc)
  cc <- computeCommunProb(cc)
  cc <- filterCommunication(cc, min.cells = 10)

  cat("  Done:", cond, "\n")
  return(cc)
})

cellchat_list <- Filter(Negate(is.null), cellchat_list)
names(cellchat_list) <- conds

# --------------------- Section 7: Top Pathways in ILD ------------------------
comm_ild <- subsetCommunication(cellchat_list[["ILD"]])

top_pathways <- comm_ild %>%
  arrange(desc(prob)) %>%
  pull(pathway_name) %>%
  unique() %>%
  head(5)

# --------------------- Section 8: Chord Diagram — Top Pathways ----------------

  "Figure_CellChat_Chord_ILD_TopPathways_Subtypes.pdf",
  width  = 11,
  height = 11
)

  cellchat_list[["ILD"]],
  signaling     = top_pathways,
  layout        = "chord",

  edge.width.max = 10,

  title.name    = "Top signaling pathways in ILD (subtypes)"

# --------------------- Section 9: Hub Gene-Mediated LR Extraction -------------
hub_genes <- c(
  "CBS", "COL3A1", "EGFR", "IL6",
  "VEGFA", "MME", "MMP1", "SERPINE1", "TTR"
)

# Query CellChatDB for hub gene-containing interactions
db <- CellChatDB.human$interaction

hub_lr_db <- db %>%
  filter(
    grepl(paste(hub_genes, collapse = "|"), ligand.symbol) |
      grepl(paste(hub_genes, collapse = "|"), receptor.symbol)
  )

cat("Hub gene-containing LR pairs in database:", nrow(hub_lr_db), "\n")

# Map hub LR pairs to ILD communication
hub_comm <- comm_ild %>%
  filter(interaction_name %in% hub_lr_db$interaction_name) %>%
  arrange(desc(prob))

cat("Top hub-mediated interactions in ILD:\n")
print(head(hub_comm, 20))

cat("Pathways involved:\n")
print(table(hub_comm$pathway_name))

cat("Cell-cell pairs involved:\n")
print(table(hub_comm$source, hub_comm$target))

# --------------------- Section 12: Target Cell Communication -----------------

# Incoming signals to target cells (all-cell communication)

  filter(target %in% target_cells) %>%

  dplyr::select(
    Receiver              = target,
    Sender                = source,
    Ligand                = ligand,
    Receptor              = receptor,
    Pathway               = pathway_name,
    `Communication prob`  = prob,
    `Interaction name`    = interaction_name
  )
